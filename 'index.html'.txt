    function readJson(key, fallback) {
      try {
        const raw = localStorage.getItem(key);
        if (!raw) return fallback;
        const parsed = JSON.parse(raw);
        return parsed && Array.isArray(parsed) ? parsed : fallback;
      } catch (error) {
        console.warn('Unable to read storage for', key, error);
        return fallback;
      }
    }

    function persistJson(key, value) {
      try {
        localStorage.setItem(key, JSON.stringify(value));
      } catch (error) {
        console.warn('Unable to write storage for', key, error);
      }
    }

    function escapeHTML(value = '') {
      return String(value).replace(/[&<>"']/g, (char) => ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#039;'
      }[char]));
    }

    function showStatus(elementId, message, type = 'info') {
      const el = document.getElementById(elementId);
      if (!el) return;
      el.textContent = message;
      el.className = `status-message show ${type}`;
      clearTimeout(el.timer);
      el.timer = setTimeout(() => {
        el.className = 'status-message';
        el.textContent = '';
      }, 3500);
    }

    function setElementText(id, value) {
      const el = document.getElementById(id);
      if (el) el.textContent = value;
    }

    function showView(viewName) {
      const views = ['view-login', 'view-user', 'view-admin'];
      for (const view of views) {
        const el = document.getElementById(view);
        if (el) el.classList.toggle('hidden', view !== viewName);
      }
    }

    function getTranslatedSection(section, lang = currentLang) {
      const key = SECTION_KEY_MAP[section] || null;
      if (!key) return section;
      return langData[lang]?.[key] || section;
    }

    function updatePlaceholders(lang) {
      const ph = placeholderMap[lang] || placeholderMap.en;
      const loginUsername = document.getElementById('login-username');
      const formTemp = document.getElementById('form-temperature');
      const formNotes = document.getElementById('form-notes');
      const newUsername = document.getElementById('new-username');
      const newPassword = document.getElementById('new-password');

      if (loginUsername) loginUsername.placeholder = ph.username;
      if (formTemp) formTemp.placeholder = ph.temp;
      if (formNotes) formNotes.placeholder = ph.notes;
      if (newUsername) newUsername.placeholder = ph.uName;
      if (newPassword) newPassword.placeholder = ph.uPass;
    }

    function changeLanguage(lang) {
      currentLang = lang;
      document.body.className = (lang === 'ar' || lang === 'ur') ? 'rtl' : 'ltr';

      document.querySelectorAll('[data-key]').forEach((element) => {
        const key = element.getAttribute('data-key');
        if (langData[lang] && langData[lang][key]) {
          element.textContent = langData[lang][key];
        }
      });

      updatePlaceholders(lang);

      if (currentUser && currentUser.role !== 'admin') {
        const label = getTranslatedSection(currentUser.section, lang);
        setElementText('user-section-badge', `${langData[lang]?.section_label || 'Wing'}: ${label.toUpperCase()}`);
        setElementText('user-form-title', `${label.toUpperCase()} ${langData[lang]?.qa_insp_sheet || 'QA Inspection Sheet'}`);
      }

      const langSelect = document.getElementById('lang-select');
      if (langSelect) langSelect.value = lang;
    }

    function handleLogin(event) {
      event.preventDefault();
      const username = document.getElementById('login-username').value.trim();
      const password = document.getElementById('login-password').value;

      if (!username || !password) {
        showStatus('login-status', langData[currentLang].required_fields, 'error');
        return;
      }

      const foundUser = appUsers.find((user) => {
        return user.username.toLowerCase() === username.toLowerCase() && user.password === password;
      });

      if (!foundUser) {
        showStatus('login-status', langData[currentLang].invalid_credentials, 'error');
        return;
      }

      currentUser = foundUser;
      document.getElementById('header-logout').classList.remove('hidden');

      if (foundUser.role === 'admin') {
        showView('view-admin');
        toggleAdminSubView('admin-reports');
      } else {
        showView('view-user');
        const inspectorInput = document.getElementById('form-inspector');
        if (inspectorInput) inspectorInput.value = foundUser.username;
        const sectionTitle = getTranslatedSection(foundUser.section, currentLang);
        setElementText('user-section-badge', `${langData[currentLang].section_label}: ${sectionTitle.toUpperCase()}`);
        const title = `${sectionTitle.toUpperCase()} ${langData[currentLang].qa_insp_sheet || 'QA Inspection Sheet'}`;
        setElementText('user-form-title', title);
      }

      document.getElementById('login-form').reset();
    }

    function logout() {
      currentUser = null;
      showView('view-login');
      document.getElementById('header-logout').classList.add('hidden');
      document.getElementById('login-form').reset();
    }

    function toggleAdminSubView(target) {
      const reportsSection = document.getElementById('subview-reports');
      const usersSection = document.getElementById('subview-users');

      if (target === 'admin-reports') {
        reportsSection.classList.remove('hidden');
        usersSection.classList.add('hidden');
        renderReports();
      } else {
        reportsSection.classList.add('hidden');
        usersSection.classList.remove('hidden');
        renderUsersList();
      }
    }

    function handleUserCreation(event) {
      event.preventDefault();
      const username = document.getElementById('new-username').value.trim();
      const password = document.getElementById('new-password').value.trim();
      const section = document.getElementById('new-section').value;

      if (!username || !password || !section) {
        showStatus('admin-status', langData[currentLang].required_fields, 'error');
        return;
      }

      if (appUsers.some((user) => user.username.toLowerCase() === username.toLowerCase())) {
        showStatus('admin-status', langData[currentLang].msg_user_exists, 'error');
        return;
      }

      appUsers.push({ username: username.toLowerCase(), password, role: 'user', section });
      persistJson(STORAGE_KEYS.users, appUsers);
      document.getElementById('user-creation-form').reset();
      renderUsersList();
      showStatus('admin-status', langData[currentLang].save_success, 'success');
    }

    function renderUsersList() {
      const container = document.getElementById('users-list-container');
      const regularUsers = appUsers.filter((user) => user.role !== 'admin');

      if (!regularUsers.length) {
        container.innerHTML = `<div class="empty-state">${langData[currentLang].no_personnel}</div>`;
        return;
      }

      container.innerHTML = regularUsers.map((user) => `
        <div class="user-item">
          <span>User: <strong>${escapeHTML(user.username)}</strong></span>
          <span class="section-badge">${escapeHTML(getTranslatedSection(user.section, currentLang))}</span>
        </div>
      `).join('');
    }

    function validateTemperature(value) {
      if (!value || value.toLowerCase() === 'n/a') return true;
      const temp = Number(value);
      return !Number.isNaN(temp);
    }

    function resizeImage(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => {
          const img = new Image();
          img.onload = () => {
            const canvas = document.createElement('canvas');
            const MAX_DIMENSION = 1200;
            let { width, height } = img;

            if (width > height && width > MAX_DIMENSION) {
              height = Math.round((height * MAX_DIMENSION) / width);
              width = MAX_DIMENSION;
            } else if (height > MAX_DIMENSION) {
              width = Math.round((width * MAX_DIMENSION) / height);
              height = MAX_DIMENSION;
            }

            canvas.width = width;
            canvas.height = height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, width, height);
            resolve(canvas.toDataURL('image/jpeg', 0.8));
          };
          img.onerror = reject;
          img.src = reader.result;
        };
        reader.onerror = reject;
        reader.readAsDataURL(file);
      });
    }

    async function handleFormSubmit(event) {
      event.preventDefault();

      const temperature = document.getElementById('form-temperature').value.trim();
      const status = document.getElementById('form-checklist').value;
      const notes = document.getElementById('form-notes').value.trim();
      const photoFile = document.getElementById('form-photo').files[0];

      if (!status || !notes || !photoFile) {
        showStatus('user-status', langData[currentLang].required_fields, 'error');
        return;
      }

      if (!validateTemperature(temperature)) {
        showStatus('user-status', langData[currentLang].invalid_temp, 'error');
        return;
      }

      if (notes.length < 5) {
        showStatus('user-status', langData[currentLang].note_length, 'error');
        return;
      }

      try {
        const image = await resizeImage(photoFile);
        const report = {
          id: Date.now(),
          inspector: currentUser.username,
          section: currentUser.section,
          temperature: temperature || 'N/A',
          status,
          notes,
          image,
          timestamp: new Date().toISOString()
        };

        auditReports.unshift(report);
        persistJson(STORAGE_KEYS.reports, auditReports);

        document.getElementById('monitoring-form').reset();
        document.getElementById('form-inspector').value = currentUser.username;
        showStatus('user-status', langData[currentLang].alert_success, 'success');
      } catch (error) {
        console.error(error);
        showStatus('user-status', 'Unable to process the evidence image.', 'error');
      }
    }

    function renderReports() {
      const feed = document.getElementById('reports-feed');
      const filterValue = document.getElementById('filter-section').value;
      const filtered = filterValue === 'ALL'
        ? auditReports
        : auditReports.filter((report) => report.section === filterValue);

      if (!filtered.length) {
        feed.innerHTML = `<div class="empty-state">${langData[currentLang].no_reports}</div>`;
        return;
      }

      feed.innerHTML = filtered.map((report) => {
        const formattedDate = new Date(report.timestamp || Date.now()).toLocaleString();
        const sectionLabel = getTranslatedSection(report.section, currentLang);
        return `
          <div class="report-card">
            <div class="meta">
              <span><strong>${escapeHTML(langData[currentLang].section_label || 'Wing')}:</strong> ${escapeHTML(sectionLabel)}</span>
              <span>${escapeHTML(formattedDate)}</span>
            </div>
            <p><strong>Inspector:</strong> ${escapeHTML(report.inspector)}</p>
            <p><strong>Temperature:</strong> ${escapeHTML(report.temperature)}</p>
            <p><strong>Status:</strong> ${escapeHTML(report.status)}</p>
            <p><strong>Notes:</strong> ${escapeHTML(report.notes)}</p>
            <img src="${report.image}" alt="Evidence for ${escapeHTML(report.section)}" />
          </div>
        `;
      }).join('');
    }

    function initEvents() {
      document.getElementById('lang-select').addEventListener('change', (event) => {
        changeLanguage(event.target.value);
      });

      document.getElementById('login-form').addEventListener('submit', handleLogin);
      document.getElementById('user-creation-form').addEventListener('submit', handleUserCreation);
      document.getElementById('monitoring-form').addEventListener('submit', handleFormSubmit);
      document.getElementById('header-logout').addEventListener('click', logout);
    }

    function initialize() {
      if ('serviceWorker' in navigator) {
        window.addEventListener('load', () => {
          navigator.serviceWorker.register('./sw.js').catch((error) => console.log('SW registration failed', error));
        });
      }

      initEvents();
      renderUsersList();
      renderReports();
      changeLanguage('en');
      showView('view-login');
    }

    initialize();
  </script>
