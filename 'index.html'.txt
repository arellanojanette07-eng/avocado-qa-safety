    let currentLang = 'en';

    function changeLanguage(lang) {
        currentLang = lang;
        document.body.className = (lang === 'ar' || lang === 'ur') ? 'rtl' : 'ltr';
        document.querySelectorAll('[data-key]').forEach(elem => {
            const key = elem.getAttribute('data-key');
            if (langData[lang][key]) { elem.innerText = langData[lang][key]; }
        });
        updatePlaceholders(lang);
        if (currentUser) {
            if (currentUser.role === 'admin') {
                renderReports(); renderUsersList();
            } else {
                const sectionTranslated = langData[lang][`wing_${currentUser.section.replace(/ and | & /g, '_').replace(/ /g, '_')}`] || currentUser.section;
                document.getElementById('user-section-badge').innerText = `${lang === 'ar' || lang === 'ur' ? 'القسم' : 'WING'}: ${sectionTranslated.toUpperCase()}`;
                document.getElementById('user-form-title').innerText = `${sectionTranslated.toUpperCase()} ${langData[lang]['qa_insp_sheet']}`;
            }
        }
    }

    let appUsers = JSON.parse(localStorage.getItem('avocado_users')) || [
        { username: 'admin', password: '123', role: 'admin', section: 'all' },
        { username: 'hotsection_user', password: '123', role: 'user', section: 'hot section' },
        { username: 'butchery_user', password: '123', role: 'user', section: 'butchery' }
    ];
    let auditReports = JSON.parse(localStorage.getItem('avocado_reports')) || [];
    let currentUser = null;

    if(!localStorage.getItem('avocado_users')) { localStorage.setItem('avocado_users', JSON.stringify(appUsers)); }

    function handleLogin(e) {
        e.preventDefault();
        const userIn = document.getElementById('login-username').value.trim();
        const passIn = document.getElementById('login-password').value;
        const foundUser = appUsers.find(u => u.username.toLowerCase() === userIn.toLowerCase() && u.password === passIn);
        if(foundUser) {
            currentUser = foundUser;
            document.getElementById('view-login').classList.add('hidden');
            document.getElementById('header-logout').classList.remove('hidden');
            if(foundUser.role === 'admin') {
                document.getElementById('view-admin').classList.remove('hidden');
                toggleAdminSubView('admin-reports');
            } else {
                document.getElementById('view-user').classList.remove('hidden');
                changeLanguage(currentLang);
                document.getElementById('form-inspector').value = foundUser.username;
            }
        } else { alert('Invalid Credentials.'); }
    }

    function logout() {
        currentUser = null;
        document.getElementById('view-admin').classList.add('hidden');
        document.getElementById('view-user').classList.add('hidden');
        document.getElementById('header-logout').classList.add('hidden');
        document.getElementById('view-login').classList.remove('hidden');
    }

    function toggleAdminSubView(target) {
        if(target === 'admin-reports') {
            document.getElementById('subview-reports').classList.remove('hidden');
            document.getElementById('subview-users').classList.add('hidden');
            renderReports();
        } else {
            document.getElementById('subview-reports').classList.add('hidden');
            document.getElementById('subview-users').classList.remove('hidden');
            renderUsersList();
        }
    }

    function handleUserCreation(e) {
        e.preventDefault();
        const uName = document.getElementById('new-username').value.trim().toLowerCase();
        const uPass = document.getElementById('new-password').value;
        const uSec = document.getElementById('new-section').value;
        if(appUsers.some(u => u.username === uName)) { alert('Exists.'); return; }
        appUsers.push({ username: uName, password: uPass, role: 'user', section: uSec });
        localStorage.setItem('avocado_users', JSON.stringify(appUsers));
        document.getElementById('new-username').value = '';
        document.getElementById('new-password').value = '';
        renderUsersList();
    }

    function renderUsersList() {
        const container = document.getElementById('users-list-container');
        const regularUsers = appUsers.filter(u => u.role !== 'admin');
        if(regularUsers.length === 0) { container.innerHTML = '<p>No personnel.</p>'; return; }
        container.innerHTML = regularUsers.map(u => `
            <div class="user-item">
                <span>User: <strong>${u.username}</strong></span>
                <span class="section-badge">${u.section}</span>
            </div>
        `).join('');
    }

    function handleFormSubmit(e) {
        e.preventDefault();
        const temp = document.getElementById('form-temperature').value || 'N/A';
        const status = document.getElementById('form-checklist').value;
        const notes = document.getElementById('form-notes').value;
        const photoFile = document.getElementById('form-photo').files[0];
        if(!photoFile) return;
        const reader = new FileReader();
        reader.onloadend = function() {
            const report = { id: Date.now(), inspector: currentUser.username, section: currentUser.section, temperature: temp, status: status, notes: notes, image: reader.result, timestamp: new Date().toLocaleString() };
            auditReports.unshift(report);
            localStorage.setItem('avocado_reports', JSON.stringify(auditReports));
            alert(langData[currentLang]['alert_success']);
            document.getElementById('monitoring-form').reset();
            document.getElementById('form-inspector').value = currentUser.username;
        }
        reader.readAsDataURL(photoFile);
    }

    function renderReports() {
        const feed = document.getElementById('reports-feed');
        const filterValue = document.getElementById('filter-section').value;
        const filtered = filterValue === 'ALL' ? auditReports : auditReports.filter(r => r.section === filterValue);
        if(filtered.length === 0) { feed.innerHTML = '<p>No safety logs.</p>'; return; }
        feed.innerHTML = filtered.map(r => `
            <div class="report-card">
                <div><strong>Sector:</strong> ${r.section.toUpperCase()}</div>
                <div><strong>Notes:</strong> ${r.notes}</div>
            </div>
        `).join('');
    }
    if ('serviceWorker' in navigator) {
        window.addEventListener('load', () => {
            navigator.serviceWorker.register('sw.js').catch(err => console.log(err));
        });
    }
    changeLanguage('en');
